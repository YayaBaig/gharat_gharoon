نمونه‌های اجرا — nobitex_trading
================================

# (اختیاری) توکن API
export NOBITEX_TOKEN="your_token_here"

# ---------------------------------------------------------------
# داشبورد وب
# ---------------------------------------------------------------
python app.py                  # پورت پیش‌فرض 8000
python app.py --port 9000

# فراخوانی مستقیم API داشبورد (وقتی سرور در حال اجراست):
curl http://localhost:8000/api/status
curl http://localhost:8000/api/symbols
curl "http://localhost:8000/api/ohlc?symbol=BTCUSDT&resolution=60"

curl -X POST http://localhost:8000/api/scan \
  -H "Content-Type: application/json" \
  -d '{"symbols":["BTCUSDT","ETHUSDT"],"capital":1000,"taker_fee":0.001,"slippage":0.002}'

# با output_dir، فایل‌های CSV و JSON هم در آن پوشه ذخیره می‌شوند:
curl -X POST http://localhost:8000/api/save \
  -H "Content-Type: application/json" \
  -d '{"symbols":["BTCUSDT"],"capital":1000,"output_dir":"output"}'
